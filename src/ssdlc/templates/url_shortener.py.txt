"""Deterministic local URL-shortener prototype; not a production web server."""

import argparse
import json
import re
import sqlite3
import time
from datetime import datetime, timezone
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import urlsplit

DAILY_ANALYTICS = __DAILY_ANALYTICS__
EXPIRATION_POLICY = __EXPIRATION_POLICY__


class Store:
    def __init__(self, path, clock=time.time):
        self.clock = clock
        self.connection = sqlite3.connect(path, check_same_thread=False)
        self.connection.execute(
            "CREATE TABLE IF NOT EXISTS links (id INTEGER PRIMARY KEY AUTOINCREMENT, "
            "url TEXT NOT NULL, clicks INTEGER NOT NULL DEFAULT 0)"
        )
        if DAILY_ANALYTICS:
            self.connection.execute(
                "CREATE TABLE IF NOT EXISTS daily_clicks (link_id INTEGER NOT NULL, "
                "day TEXT NOT NULL, clicks INTEGER NOT NULL, PRIMARY KEY(link_id, day))"
            )
        if EXPIRATION_POLICY:
            self.connection.execute(
                "CREATE TABLE IF NOT EXISTS expiration (link_id INTEGER PRIMARY KEY, "
                "expires_at REAL NOT NULL)"
            )
        self.connection.commit()

    def create(self, url, ttl=None):
        if not isinstance(url, str) or len(url) > 2048 or any(ord(c) < 33 for c in url):
            raise ValueError("Invalid URL")
        try:
            target = urlsplit(url)
            valid = target.scheme in {"http", "https"} and target.hostname
            valid = valid and not target.username and not target.password
            _port = target.port
        except ValueError:
            valid = False
        if not valid:
            raise ValueError("HTTP or HTTPS URL required")
        if EXPIRATION_POLICY:
            ttl = EXPIRATION_POLICY["default_ttl"] if ttl is None else ttl
            if ttl is None and EXPIRATION_POLICY["mandatory"]:
                raise ValueError("TTL is required")
            if ttl is not None and (type(ttl) is not int or ttl <= 0 or ttl > 31536000):
                raise ValueError("TTL must be positive integer seconds, at most one year")
        elif ttl is not None:
            raise ValueError("Expiration is not supported")
        with self.connection:
            cursor = self.connection.execute("INSERT INTO links(url) VALUES (?)", (url,))
            identifier = cursor.lastrowid
            if EXPIRATION_POLICY and ttl is not None:
                self.connection.execute(
                    "INSERT INTO expiration VALUES (?, ?)", (identifier, self.clock() + ttl)
                )
        return format(identifier, "x")

    def resolve(self, code):
        identifier = self._identifier(code)
        if identifier is None:
            return 404, None
        with self.connection:
            row = self.connection.execute("SELECT url FROM links WHERE id=?", (identifier,)).fetchone()
            if row is None:
                return 404, None
            if EXPIRATION_POLICY:
                expiry = self.connection.execute(
                    "SELECT expires_at FROM expiration WHERE link_id=?", (identifier,)
                ).fetchone()
                if expiry and self.clock() >= expiry[0]:
                    if EXPIRATION_POLICY["delete_on_access"]:
                        self.connection.execute("DELETE FROM links WHERE id=?", (identifier,))
                        self.connection.execute("DELETE FROM expiration WHERE link_id=?", (identifier,))
                    return EXPIRATION_POLICY["expired_status"], None
            self.connection.execute("UPDATE links SET clicks=clicks+1 WHERE id=?", (identifier,))
            if DAILY_ANALYTICS:
                day = datetime.fromtimestamp(self.clock(), timezone.utc).date().isoformat()
                self.connection.execute(
                    "INSERT INTO daily_clicks VALUES (?, ?, 1) "
                    "ON CONFLICT(link_id, day) DO UPDATE SET clicks=clicks+1", (identifier, day)
                )
        return 302, row[0]

    def analytics(self, code):
        identifier = self._identifier(code)
        if identifier is None:
            return None
        row = self.connection.execute("SELECT clicks FROM links WHERE id=?", (identifier,)).fetchone()
        if row is None:
            return None
        result = {"code": code, "clicks": row[0]}
        if DAILY_ANALYTICS:
            result["daily"] = dict(self.connection.execute(
                "SELECT day, clicks FROM daily_clicks WHERE link_id=? ORDER BY day", (identifier,)
            ).fetchall())
        return result

    def close(self):
        self.connection.close()

    @staticmethod
    def _identifier(code):
        if not isinstance(code, str) or not re.fullmatch(r"[1-9a-f][0-9a-f]{0,14}", code):
            return None
        return int(code, 16)


def create_server(database="links.sqlite", host="127.0.0.1", port=8000):
    store = Store(database)

    class Handler(BaseHTTPRequestHandler):
        def log_message(self, format, *args):
            # Avoid recording targets, client addresses or request paths.
            pass

        def _reply(self, status, payload, headers=None):
            body = json.dumps(payload).encode("utf-8")
            self.send_response(status)
            self.send_header("Content-Type", "application/json")
            self.send_header("Content-Length", str(len(body)))
            self.send_header("Cache-Control", "no-store")
            for key, value in (headers or {}).items():
                self.send_header(key, value)
            self.end_headers()
            self.wfile.write(body)

        def _handle(self):
            path = urlsplit(self.path).path
            if path == "/health" and self.command == "GET":
                self._reply(200, {"status": "ok"})
            elif path == "/links" and self.command == "POST":
                try:
                    size = int(self.headers.get("Content-Length", "0"))
                except ValueError:
                    self._reply(400, {"error": "INVALID_REQUEST"})
                    return
                if size < 1 or size > 8192:
                    self._reply(413 if size > 8192 else 400, {"error": "INVALID_BODY_SIZE"})
                    return
                body = self.rfile.read(size)
                if self.headers.get("Content-Type", "").split(";")[0].strip() != "application/json":
                    self._reply(415, {"error": "JSON_REQUIRED"})
                    return
                try:
                    request = json.loads(body)
                    allowed = {"url", "ttl_seconds"} if EXPIRATION_POLICY else {"url"}
                    if not isinstance(request, dict) or "url" not in request or set(request) - allowed:
                        raise ValueError("Invalid fields")
                    code = store.create(request["url"], request.get("ttl_seconds"))
                except (ValueError, UnicodeError):
                    self._reply(400, {"error": "INVALID_REQUEST"})
                    return
                self._reply(201, {"code": code, "short_url": f"/r/{code}"})
            elif self.command == "GET" and re.fullmatch(r"/r/[^/]+", path):
                code = path.split("/")[-1]
                status, target = store.resolve(code)
                if target is None:
                    self._reply(status, {"error": "EXPIRED" if status == 410 else "NOT_FOUND"})
                else:
                    self._reply(302, {}, {"Location": target})
            elif self.command == "GET" and re.fullmatch(r"/links/[^/]+/analytics", path):
                result = store.analytics(path.split("/")[2])
                self._reply(200 if result is not None else 404, result or {"error": "NOT_FOUND"})
            else:
                self._reply(404, {"error": "NOT_FOUND"})

        def do_GET(self):
            try:
                self._handle()
            except sqlite3.Error:
                self._reply(503, {"error": "STORE_UNAVAILABLE"})
            except Exception:  # noqa: BLE001 - sanitize unexpected request failures
                self._reply(500, {"error": "INTERNAL_ERROR"})

        do_POST = do_GET

    try:
        server = HTTPServer((host, port), Handler)
    except Exception:
        store.close()
        raise
    server.store = store
    return server


def main():
    parser = argparse.ArgumentParser(description="Local URL-shortener prototype")
    parser.add_argument("--database", default="links.sqlite")
    parser.add_argument("--port", type=int, default=8000)
    args = parser.parse_args()
    server = create_server(args.database, port=args.port)
    print(f"Listening on http://127.0.0.1:{server.server_port}", flush=True)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()
        server.store.close()


if __name__ == "__main__":
    main()
